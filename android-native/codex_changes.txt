+package com.example.wallet

+import android.util.Base64
+import kotlinx.coroutines.Dispatchers
+import kotlinx.coroutines.withContext
+import org.json.JSONArray
+import org.json.JSONObject
+import java.io.IOException
+import java.net.HttpURLConnection
+import java.net.URL
+
+/** Transport for the loopback Devnet API exposed to the device by adb reverse. */
+internal class DevnetShopClient {
+    data class Challenge(val nonce: String, val message: String)
+
+    suspend fun challenge(wallet: String): Challenge {
+        val response = post("/auth/challenge", JSONObject().put("wallet", wallet))
+        return Challenge(response.getString("nonce"), response.getString("message"))
+    }
+
+    suspend fun authenticate(challenge: Challenge, signature: ByteArray): String =
+        post(
+            "/auth/session",
+            JSONObject().put("nonce", challenge.nonce)
+                .put("signature", Base64.encodeToString(signature, Base64.NO_WRAP))
+        ).getString("token")
+
+    suspend fun createOrder(token: String): JSONObject = post("/orders/create", token = token)
+
+    suspend fun orders(token: String): JSONArray =
+        post("/orders/list", token = token).getJSONArray("orders")
+
+    suspend fun preparePayment(token: String, orderId: String): JSONObject =
+        post("/orders/payment", JSONObject().put("orderId", orderId), token)
+
+    suspend fun recover(token: String, orderId: String): JSONObject =
+        post("/orders/recover", JSONObject().put("orderId", orderId), token)
+
+    suspend fun broadcast(token: String, orderId: String, signedTransaction: String): String =
+        post("/orders/broadcast", JSONObject().put("orderId", orderId)
+            .put("transaction", signedTransaction), token).getString("signature")
+
+    suspend fun settle(token: String, orderId: String, signature: String): JSONObject =
+        post("/orders/settle", JSONObject().put("orderId", orderId).put("signature", signature), token)
+
+    suspend fun inventory(token: String): JSONArray =
+        post("/inventory", token = token).getJSONArray("items")
+
+    private suspend fun post(path: String, body: JSONObject = JSONObject(), token: String? = null): JSONObject =
+        withContext(Dispatchers.IO) {
+            val connection = (URL("http://localhost:8765$path").openConnection() as HttpURLConnection).apply {
+                requestMethod = "POST"
+                instanceFollowRedirects = false
+                connectTimeout = 5_000
+                readTimeout = 20_000
+                doOutput = true
+                setRequestProperty("Content-Type", "application/json; charset=utf-8")
+                if (token != null) setRequestProperty("Authorization", "Bearer $token")
+            }
+            try {
+                connection.outputStream.use { it.write(body.toString().toByteArray(Charsets.UTF_8)) }
+                val status = connection.responseCode
+                if (status !in 200..299) {
+                    val detail = connection.errorStream?.bufferedReader()?.use { it.readText() }
+                    val message = detail?.let { runCatching { JSONObject(it).optString("error") }.getOrNull() }
+                    throw IOException(message?.takeIf(String::isNotBlank) ?: "Devnet shop HTTP $status")
+                }
+                connection.inputStream.bufferedReader().use { JSONObject(it.readText()) }
+            } finally {
+                connection.disconnect()
+            }
+        }
+}

+package com.example.wallet
+
+import java.math.BigInteger
+
+object SolanaAddress {
+    private const val ALPHABET = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"
+
+    fun isValid(address: String): Boolean {
+        val value = address.trim()
+        if (value.length !in 32..44 || value.any { ALPHABET.indexOf(it) < 0 }) return false
+
+        var number = BigInteger.ZERO
+        for (char in value) {
+            number = number.multiply(BigInteger.valueOf(58L))
+                .add(BigInteger.valueOf(ALPHABET.indexOf(char).toLong()))
+        }
+
+        val leadingZeroes = value.takeWhile { it == '1' }.length
+        val encodedNumber = if (number == BigInteger.ZERO) {
+            0
+        } else {
+            number.toByteArray().let { bytes ->
+                if (bytes.firstOrNull() == 0.toByte()) bytes.size - 1 else bytes.size
+            }
+        }
+        return leadingZeroes + encodedNumber == 32
+    }
+
+    fun encode(publicKey: ByteArray): String {
+        require(publicKey.size == 32) { "Solana public keys must be 32 bytes" }
+        return encodeBytes(publicKey)
+    }
+
+    internal fun encodeBytes(publicKey: ByteArray): String {
+
+        val leadingZeroes = publicKey.takeWhile { it == 0.toByte() }.size
+        var number = BigInteger(1, publicKey)
+        val base = BigInteger.valueOf(58L)
+        val encoded = StringBuilder()
+        while (number > BigInteger.ZERO) {
+            val division = number.divideAndRemainder(base)
+            encoded.append(ALPHABET[division[1].toInt()])
+            number = division[0]
+        }
+        repeat(leadingZeroes) { encoded.append('1') }
+        return encoded.reverse().toString()
+    }

+import argparse
+import base64
+import hashlib
+import json
+import secrets
+import sqlite3
+import time
+from contextlib import closing
+from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

+from cryptography.exceptions import InvalidSignature
+from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PublicKey
+
+from devnet_orders import create_order, open_database, rpc, settle
+from devnet_payment import broadcast_payment, prepare_payment

+BASE58 = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"
+AUTH_SCHEMA = """
+CREATE TABLE IF NOT EXISTS challenges (
+    nonce TEXT PRIMARY KEY, payer TEXT NOT NULL, message TEXT NOT NULL, expires INTEGER NOT NULL
+);
+CREATE TABLE IF NOT EXISTS sessions (
+    digest TEXT PRIMARY KEY, payer TEXT NOT NULL, expires INTEGER NOT NULL
+);
+"""
+
+
+def wallet_key(address):
+    if not isinstance(address, str) or not 32 <= len(address) <= 44:
+        raise ValueError("Invalid wallet address")
+    number = 0
+    for char in address:
+        number = number * 58 + BASE58.index(char)
+    key = bytes(len(address) - len(address.lstrip("1")))
+    key += number.to_bytes((number.bit_length() + 7) // 8, "big")
+    return Ed25519PublicKey.from_public_bytes(key)
+
+
+def digest(token):
+    return hashlib.sha256(token.encode()).hexdigest()
+
+
+class ApiHandler(BaseHTTPRequestHandler):
+    def respond(self, status, data):
+        output = json.dumps(data).encode()
+        self.send_response(status)
+        self.send_header("Content-Type", "application/json")
+        self.send_header("Cache-Control", "no-store")
+        self.send_header("Content-Length", str(len(output)))
+        self.end_headers()
+        self.wfile.write(output)
+
+    def do_POST(self):
+        try:
+            length = int(self.headers.get("Content-Length", "0"))
+            if not 0 < length <= 4096:
+                raise ValueError("Expected a JSON object up to 4096 bytes")
+            body = json.loads(self.rfile.read(length))
+            if not isinstance(body, dict):
+                raise ValueError("Expected a JSON object")
+            with closing(open_database(self.server.database)) as db:
+                status, result = self.dispatch(db, body)
+            self.respond(status, result)
+        except (ValueError, KeyError, TypeError, InvalidSignature) as error:
+            self.respond(400, {"error": str(error) or "Invalid signature"})
+        except sqlite3.IntegrityError:
+            self.respond(409, {"error": "Payment or entitlement already recorded"})
+        except (OSError, sqlite3.Error):
+            self.respond(503, {"error": "Devnet verification temporarily unavailable"})
+
+    def dispatch(self, db, body):
+        now = int(time.time())
+        if self.path == "/auth/challenge":
+            payer = body["wallet"]
+            wallet_key(payer)
+            nonce = secrets.token_urlsafe(24)
+            expires = now + 300
+            message = (
+                "Sprite Bonk Devnet shop login\n"
+                "Scope: local Devnet orders and inventory. No transaction approval.\n"
+                f"Wallet: {payer}\nNonce: {nonce}\nExpires (Unix UTC): {expires}"
+            )
+            with db:
+                db.execute("DELETE FROM challenges WHERE expires <= ?", (now,))
+                db.execute("INSERT INTO challenges VALUES (?, ?, ?, ?)", (nonce, payer, message, expires))
+            return 200, {"nonce": nonce, "message": message, "expires": expires}
+
+        if self.path == "/auth/session":
+            challenge = db.execute("SELECT * FROM challenges WHERE nonce=? AND expires>?",
+                                   (body["nonce"], now)).fetchone()
+            if challenge is None:
+                return 401, {"error": "Challenge expired or already used"}
+            signature = base64.b64decode(body["signature"], validate=True)
+            wallet_key(challenge["payer"]).verify(signature, challenge["message"].encode())
+            token = secrets.token_urlsafe(32)
+            with db:
+                consumed = db.execute("DELETE FROM challenges WHERE nonce=? AND expires>?",
+                                      (challenge["nonce"], int(time.time())))
+                if consumed.rowcount != 1:
+                    return 401, {"error": "Challenge expired or already used"}
+                db.execute("DELETE FROM sessions WHERE expires <= ?", (now,))
+                db.execute("INSERT INTO sessions VALUES (?, ?, ?)", (digest(token), challenge["payer"], now + 3600))
+            return 200, {"token": token, "expires": now + 3600}
+
+        scheme, _, token = self.headers.get("Authorization", "").partition(" ")
+        session = db.execute("SELECT payer FROM sessions WHERE digest=? AND expires>?",
+                             (digest(token), now)).fetchone()
+        if scheme != "Bearer" or session is None:
+            return 401, {"error": "Wallet login required"}
+        payer = session["payer"]
+        if self.path == "/orders/create":
+            return 201, create_order(db, payer)
+        if self.path == "/orders/list":
+            rows = db.execute("""SELECT orders.*, EXISTS(
+                SELECT 1 FROM closed_unpaid_orders WHERE order_id=orders.id
+                ) AS closed_unpaid, (orders.expires <= ? AND NOT EXISTS(
+                SELECT 1 FROM payment_attempts WHERE order_id=orders.id
+                )) AS expired_without_payment
+                FROM orders WHERE payer=? ORDER BY created DESC, id DESC""", (now, payer))
+            return 200, {"orders": [dict(row) for row in rows]}
+        if self.path == "/orders/payment":
+            order = db.execute("SELECT * FROM orders WHERE id=? AND payer=?",
+                               (body["orderId"], payer)).fetchone()
+            if order is None:
+                return 404, {"error": "Unknown order"}
+            if db.execute("SELECT 1 FROM closed_unpaid_orders WHERE order_id=?", (order["id"],)).fetchone():
+                return 409, {"error": "订单已关闭"}
+            return 200, prepare_payment(db, order)
+        if self.path == "/orders/settle":
+            order_id = body["orderId"]
+            if db.execute("SELECT 1 FROM orders WHERE id=? AND payer=?", (order_id, payer)).fetchone() is None:
+                return 404, {"error": "Unknown order"}
+            return 200, settle(db, order_id, body["signature"])
+        if self.path == "/orders/broadcast":
+            order = db.execute("SELECT * FROM orders WHERE id=? AND payer=?",
+                               (body["orderId"], payer)).fetchone()
+            if order is None:
+                return 404, {"error": "Unknown order"}
+            if db.execute("SELECT 1 FROM closed_unpaid_orders WHERE order_id=?", (order["id"],)).fetchone():
+                return 409, {"error": "订单已关闭"}
+            return 200, broadcast_payment(db, order, body["transaction"])
+        if self.path == "/orders/recover":
+            order_id = body["orderId"]
+            order = db.execute("SELECT * FROM orders WHERE id=? AND payer=?", (order_id, payer)).fetchone()
+            if order is None:
+                return 404, {"error": "Unknown order"}
+            if order["signature"]:
+                return 200, {"status": "fulfilled", "order": order_id}
+            attempt = db.execute("SELECT last_height FROM payment_attempts WHERE order_id=?",
+                                 (order_id,)).fetchone()
+            expired = attempt is not None and rpc("getBlockHeight", [
+                {"commitment": "finalized"},
+            ]) > attempt["last_height"]
+            entries = rpc("getSignaturesForAddress", [
+                order["destination"], {"limit": 100, "commitment": "finalized"},
+            ])
+            for entry in entries:
+                memo = entry.get("memo") or ""
+                # RPC prefixes displayed memos with their length, e.g. "[32] …".
+                if entry["err"] is None and order_id in memo:
+                    try:
+                        return 200, settle(db, order_id, entry["signature"])
+                    except ValueError:
+                        continue
+            if expired and len(entries) < 100 and not any(
+                entry["err"] is None and order_id in (entry.get("memo") or "") for entry in entries
+            ):
+                with db:
+                    db.execute("INSERT OR IGNORE INTO closed_unpaid_orders VALUES (?)", (order_id,))
+                return 200, {"status": "closed_unpaid", "order": order_id}
+            return 200, {"status": "not_found", "order": order_id}
+        if self.path == "/inventory":
+            rows = db.execute("SELECT product, order_id FROM inventory WHERE payer=?", (payer,))
+            return 200, {"items": [dict(row) for row in rows]}
+        return 404, {"error": "Unknown endpoint"}
+
+
+def main():
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("--db", required=True)
+    parser.add_argument("--port", default=8765, type=int)
+    args = parser.parse_args()
+    with closing(open_database(args.db)) as db:
+        db.executescript(AUTH_SCHEMA)
+    with ThreadingHTTPServer(("127.0.0.1", args.port), ApiHandler) as server:
+        server.database = args.db
+        server.serve_forever()
+

+if __name__ == "__main__":
+    main()

+def verify_payment(order, tx):
+    meta = tx["meta"]
+    message = tx["transaction"]["message"]
+    keys = message["accountKeys"]
+    if meta is None or meta["err"] is not None:
+        raise ValueError("Transaction failed")
+    if not order["created"] <= (tx.get("blockTime") or 0) <= order["expires"]:
+        raise ValueError("Payment landed outside the order window")
+    if not any(key["signer"] and key["pubkey"] == order["payer"] for key in keys):
+        raise ValueError("Expected payer did not sign")

+    if not any(ix.get("programId") == MEMO_PROGRAM and ix.get("parsed") == order["id"]
+               for ix in instructions):
+        raise ValueError("Order memo does not match")
+    transfers = []
+    for ix in instructions:
+        parsed = ix.get("parsed")
+        if not isinstance(parsed, dict) or parsed.get("type") != "transferChecked":
+            continue
+        info = parsed["info"]
+        if ix.get("program") in ("spl-token", "spl-token-2022") and info.get("destination") == order["destination"]:
+            transfers.append(info)
+    if len(transfers) != 1:
+        raise ValueError("Expected exactly one checked token transfer to the merchant")
+    transfer = transfers[0]
+    if (transfer["mint"] != order["mint"] or transfer["authority"] != order["payer"]
+            or int(transfer["tokenAmount"]["amount"]) != order["amount"]):
+        raise ValueError("Invalid mint, transfer authority, or amount")
+
+    def balance(entries):
+        return sum(int(entry["uiTokenAmount"]["amount"]) for entry in entries
+                   if entry["mint"] == order["mint"]
+                   and keys[entry["accountIndex"]]["pubkey"] == order["destination"])
+
+    if balance(meta.get("postTokenBalances", [])) - balance(meta.get("preTokenBalances", [])) != order["amount"]:
+        raise ValueError("Merchant balance change does not match the order")
+

+    order = db.execute("SELECT * FROM orders WHERE id=?", (order_id,)).fetchone()
+    if order is None:
+        raise ValueError("Unknown order")
+    if order["signature"] == signature:
+        return {"status": "fulfilled", "order": order_id}
+    tx = rpc("getTransaction", [signature, {
+        "encoding": "jsonParsed", "commitment": "finalized", "maxSupportedTransactionVersion": 0,
+    }])
+    if tx is None:
+        return {"status": "awaiting_finality", "order": order_id}
+    verify_payment(order, tx)
+    with db:
+        updated = db.execute("UPDATE orders SET signature=? WHERE id=? AND signature IS NULL",
+                             (signature, order_id))
+        if updated.rowcount == 0:
+            existing = db.execute("SELECT signature FROM orders WHERE id=?", (order_id,)).fetchone()
+            if existing["signature"] != signature:
+                raise ValueError("Order already paid with a different transaction")
+        else:
+            db.execute("INSERT INTO inventory VALUES (?, ?, ?)",
+                       (order["payer"], order["product"], order_id))
+    return {"status": "fulfilled", "order": order_id}
+

+"""Build a single, reusable unsigned payment attempt for a Devnet order."""
+
+import base64
+import logging
+import struct
+import time
+
+from devnet_orders import MEMO_PROGRAM, TOKEN_PROGRAM, rpc

+logger = logging.getLogger(__name__)
+COMPUTE_BUDGET = "ComputeBudget111111111111111111111111111111"
+MAX_PRIORITY_FEE = 10_000
+

+def account_permissions(message):
+    header = message.header
+    keys = message.account_keys
+    signers = header.num_required_signatures
+    if len(set(keys)) != len(keys):
+        raise ValueError("付款交易含重复账户")
+    return {
+        key: (
+            index < signers,
+            index < signers - header.num_readonly_signed_accounts
+            if index < signers else index < len(keys) - header.num_readonly_unsigned_accounts,
+        )
+        for index, key in enumerate(keys)
+    }
+
+
+def resolved_instructions(message):
+    keys = message.account_keys
+    return [
+        (keys[ix.program_id_index], tuple(keys[index] for index in ix.accounts), bytes(ix.data))
+        for ix in message.instructions
+    ]

+def broadcast_payment(db, order, payload):
+    from solders.pubkey import Pubkey
+    from solders.transaction import Transaction
+
+    if order["signature"]:
+        return {"signature": order["signature"]}
+    attempt = db.execute("SELECT payload FROM payment_attempts WHERE order_id=?", (order["id"],)).fetchone()
+    if attempt is None:
+        raise ValueError("订单没有付款请求")
+    try:
+        signed = Transaction.from_bytes(base64.b64decode(payload, validate=True))
+        expected = Transaction.from_bytes(base64.b64decode(attempt["payload"]))
+    except (ValueError, TypeError) as error:
+        raise ValueError("钱包返回无效交易") from error
+    signed.sanitize()
+    if signed.verify_with_results() != [True]:
+        raise ValueError("钱包返回的付款签名无法通过验证")
+    actual = signed.message
+    quoted = expected.message
+    if actual.account_keys[0] != quoted.account_keys[0]:
+        raise ValueError("钱包更换了付款账户")
+    actual_permissions = account_permissions(actual)
+    quoted_permissions = account_permissions(quoted)
+    actual_instructions = resolved_instructions(actual)
+    budget = {}
+    payment_instructions = []
+    for program, accounts, data in actual_instructions:
+        if str(program) != COMPUTE_BUDGET:
+            payment_instructions.append((program, accounts, data))
+            continue
+        # Only SetComputeUnitLimit and SetComputeUnitPrice are accepted.
+        if accounts or not data or (data[0], len(data)) not in ((2, 5), (3, 9)) or data[0] in budget:
+            raise ValueError("钱包添加了不支持的计算预算指令")
+        budget[data[0]] = int.from_bytes(data[1:], "little")
+        if actual_permissions.get(program) != (False, False):
+            raise ValueError("计算预算程序的账户权限不正确")
+    if budget:
+        del actual_permissions[Pubkey.from_string(COMPUTE_BUDGET)]
+    limit = budget.get(2, 1_400_000)
+    priority_fee = (limit * budget.get(3, 0) + 999_999) // 1_000_000
+    if not 0 < limit <= 1_400_000 or priority_fee > MAX_PRIORITY_FEE:
+        raise ValueError(f"钱包请求的优先费为 {priority_fee} lamports，超过 {MAX_PRIORITY_FEE} 上限")
+    quoted_permissions.pop(Pubkey.from_string(COMPUTE_BUDGET), None)
+    if actual_permissions != quoted_permissions:
+        actual_programs = {program for program, _, _ in resolved_instructions(actual)}
+        quoted_programs = {program for program, _, _ in resolved_instructions(quoted)}
+        logger.warning(
+            "Payment %s account mismatch: added=%d removed=%d permission_changes=%s "
+            "programs_added=%s programs_removed=%s instructions=%d vs %d blockhash_changed=%s",
+            order["id"],
+            len(actual_permissions.keys() - quoted_permissions.keys()),
+            len(quoted_permissions.keys() - actual_permissions.keys()),
+            [
+                (quoted_permissions[key], actual_permissions[key])
+                for key in quoted_permissions.keys() & actual_permissions.keys()
+                if quoted_permissions[key] != actual_permissions[key]
+            ],
+            sorted(map(str, actual_programs - quoted_programs)),
+            sorted(map(str, quoted_programs - actual_programs)),
+            len(actual.instructions), len(quoted.instructions),
+            actual.recent_blockhash != quoted.recent_blockhash,
+        )
+        raise ValueError("钱包更改了付款交易的账户配置")
+    if payment_instructions != [
+        ix for ix in resolved_instructions(quoted) if str(ix[0]) != COMPUTE_BUDGET
+    ]:
+        raise ValueError("钱包更改了付款交易的指令")
+    if actual.recent_blockhash != quoted.recent_blockhash:
+        raise ValueError("钱包更换了付款交易的区块哈希")
+    signature = str(signed.signatures[0])
+    rpc("sendTransaction", [payload, {"encoding": "base64", "preflightCommitment": "confirmed"}])
+    return {"signature": signature}

+def prepare_payment(db, order):
+    from solders.hash import Hash
+    from solders.instruction import AccountMeta, Instruction
+    from solders.message import Message
+    from solders.pubkey import Pubkey
+    from solders.transaction import Transaction
+
+    if order["signature"] or order["expires"] <= int(time.time()):
+        raise ValueError("订单已结算或已过期")
+
+    def existing():
+        return db.execute("SELECT * FROM payment_attempts WHERE order_id=?", (order["id"],)).fetchone()
+
+    def response(attempt):
+        transaction = Transaction.from_bytes(base64.b64decode(attempt["payload"]))
+        # The final instruction is the server-generated TransferChecked.
+        decimals = transaction.message.instructions[-1].data[-1]
+        return {
+            "transaction": attempt["payload"], "lastValidBlockHeight": attempt["last_height"],
+            "feeLamports": attempt["fee"], "maxFeeLamports": attempt["fee"] + MAX_PRIORITY_FEE,
+            "tokenDecimals": decimals,
+        }
+
+    attempt = existing()
+    if attempt:
+        height = rpc("getBlockHeight", [{"commitment": "confirmed"}])
+        if height > attempt["last_height"]:
+            raise ValueError("付款请求已失效，请核对原订单后再创建新订单")
+        return response(attempt)
+
+    payer = Pubkey.from_string(order["payer"])
+    mint = Pubkey.from_string(order["mint"])
+    destination = Pubkey.from_string(order["destination"])
+    accounts = rpc("getTokenAccountsByOwner", [
+        str(payer), {"mint": str(mint)}, {"encoding": "jsonParsed", "commitment": "confirmed"},
+    ])["value"]
+    source = None
+    for entry in accounts:
+        account = entry["account"]
+        info = account["data"]["parsed"]["info"]
+        amount = info["tokenAmount"]

+                and entry["pubkey"] != str(destination) and int(amount["amount"]) >= order["amount"]):
+            source = Pubkey.from_string(entry["pubkey"])
+            decimals = amount["decimals"]
+            break

+        raise ValueError("当前钱包没有足够的 Devnet 测试代币")

+    transfer = Instruction(Pubkey.from_string(TOKEN_PROGRAM), struct.pack("<BQB", 12, order["amount"], decimals), [
+        AccountMeta(source, False, True), AccountMeta(mint, False, False),
+        AccountMeta(destination, False, True), AccountMeta(payer, True, False),
+    ])
+    memo = Instruction(Pubkey.from_string(MEMO_PROGRAM), order["id"].encode("ascii"), [])
+    block = rpc("getLatestBlockhash", [{"commitment": "confirmed"}])["value"]

+
+    message = Message.new_with_blockhash(
+        [set_compute_unit_limit(25_000), set_compute_unit_price(0), memo, transfer],
+        payer, Hash.from_string(block["blockhash"]),

+    if fee is None:
+        raise ValueError("无法读取交易手续费，请重试")
+    if rpc("getBalance", [str(payer), {"commitment": "confirmed"}])["value"] < fee:
+        raise ValueError("当前钱包没有足够的 Devnet SOL 支付手续费")
+    payload = base64.b64encode(bytes(Transaction.new_unsigned(message))).decode("ascii")
+    with db:
+        db.execute("BEGIN IMMEDIATE")
+        if order["expires"] <= int(time.time()):
+            raise ValueError("订单已过期，请重新下单")
+        db.execute("INSERT OR IGNORE INTO payment_attempts VALUES (?, ?, ?, ?)",
+                   (order["id"], payload, block["lastValidBlockHeight"], fee))
+    return response(existing())
